```python
import os
import requests

api_key = os.environ["LLM_API_KEY"]
endpoint_url = "https://xxx.xxx.com/v1/chat/completions" # 注意结尾是 /v1/chat/completions


class LLMClient:
    def __init__(self, api_key, endpoint_url):
        self.endpoint_url = endpoint_url
        self.headers = {
            "Authorization": f"Bearer {api_key}",
            "Content-Type": "application/json",
        }

    def call(self, payload):
        response = requests.post(
            self.endpoint_url,
            headers=self.headers,
            json=payload,
            timeout=(5, 120),
        )
        response.raise_for_status()
        return response.json()

llm_client = LLMClient(api_key, endpoint_url)
```

**流式输出**

```python
import json
import requests

payload = {
    "model": "DeepSeek-V3.2",
    "messages": [{"role": "user", "content": "写一个Python函数，判断素数。"}],
    "stream": True,
}

with requests.post(
    base_url,
    headers={"Authorization": f"Bearer {api_key}"},
    json=payload,
    stream=True,
    timeout=(5, 120),
) as response:
    if not response.ok:
        raise RuntimeError(f"HTTP {response.status_code}: {response.text}")

    for raw_line in response.iter_lines(chunk_size=1):
        if not raw_line:
            continue

        line = raw_line.decode("utf-8")
        if not line.startswith("data:"):
            continue

        data = line[5:].strip()
        if data == "[DONE]":
            break

        chunk = json.loads(data)
        choices = chunk.get("choices") or []
        if choices:
            content = (choices[0].get("delta") or {}).get("content")
            if content:
                print(content, end="", flush=True)

print()
```

```python
tools = [{
    "type": "function",
    "function": {
        "name": "get_weather",
        "description": "查询指定城市的天气和气温",
        "parameters": {
            "type": "object",
            "properties": {
                "city": {"type": "string", "description": "城市名称"}
            },
            "required": ["city"],
        },
    },
}]

messages = [
    {"role": "user", "content": "查一下北京的天气，并给我一条穿衣建议。"}
]

for _ in range(3):
    resp = llm_client.call({
        "model": 'DeepSeek-V4-Flash-0731',
        "messages": messages,
        "tools": tools,
        "tool_choice": "auto",
    })

    message = resp["choices"][0]["message"]
    tool_calls = message.get("tool_calls") or []


    messages.append(message)

    if not tool_calls:
        print(message["content"])
        break


    for tool_call in tool_calls:
        function = tool_call["function"]
        if function["name"] != "get_weather":
            raise ValueError(f"未知工具：{function['name']}")

        arguments = json.loads(function["arguments"])
        result = get_weather(arguments["city"])

        messages.append({
            "role": "tool",
            "tool_call_id": tool_call["id"],
            "content": json.dumps(result, ensure_ascii=False),
        })
else:
    raise RuntimeError("工具调用次数过多")
```

## Agent

Agent 是包含了 RAG、提示词、Function calling、Re-Act等一系列元素的集合
- API 调用，例如查询天气、发送邮件
- Re-Act


**自主Agent**
- 由 Planner 模块（也是个LLM），与其它 LLM、工具调用的配合


Agent 优化
- GEPA (Genetic-Pareto)
    - 边界内收敛到最优
    - 优化 prompt
- Auto Research
    - 开放的边界，探索高价值改动方向
    - 优化整个工作流
- Agentic RL (Agentic Reinforcement Learning)
    - 根据完整轨迹，对模型做后训练


### context 构建

Context 包括什么
- System Prompt
    - 定义角色、能力、行为约束
- User Goal
    - 用户的原始目标
- Task State
    - 目前的执行状态
- Conversation History
    - 历史对话，它是无限增长的，因此需要 **压缩**（Summarization）
- Memory：长期应该知道什么
    - Profile Memory，用户画像、偏好等
    - Episodic Memory，历史事件。之前解决的问题、踩过的坑
    - Semantic Memory，最佳实践、python 包
- Retrieved Knowledge
    - RAG 得到的信息，例如从1000个文件中检索到的 2个相关文件
- Tool Description
    - 有哪些工具可供调用
- Tool Results
    - 工具返回结果
- Previous Reasoning
    - 上一步的推理状态

```python
tools = [
    # shell 工具，LLM 根据通用知识确定如何写 shell 命令，而不是提供大量的、地球人都知道的 Linux 命令
    {

            """,

                    "command"
                ]
            }
        }
    }

    # 自定义 python 函数，明确告诉 LLM 这个函数如何使用
    , {
        "type": "function",
        "function": {
            "name": "get_weather",
            "description": "查询指定城市的天气和气温",
            "parameters": {
                "type": "object",

                    "city": {"type": "string", "description": "城市名称"}

import subprocess
import json


def terminal(command):
    """执行 shell 命令，
    没有做安全过滤，请不要让 agent 执行危险动作"""
    result = subprocess.run(
        command,
        shell=True,
        capture_output=True,
        text=True
    )

    return result.stdout + result.stderr


def get_weather(city):
    """模拟天气 API"""
    return f"{city} 今天晴，温度 25℃"


def tool_routine(tool_call):
    """工具执行路由，agent 只调用此函数，由此函数配置具体的工具"""
    name = tool_call["function"]["name"]

    arguments = json.loads(
        tool_call["function"]["arguments"]
    )

    if name == "terminal":
        result = terminal(**arguments)

    elif name == "get_weather":
        result = get_weather(**arguments)

    else:
        result = f"Unknown tool: {name}"

    print(f"准备调用工具{name}, 其入参为 {arguments}")

    return result
```

agent：
```python
model_name = 'DeepSeek-V4.1-Flash'

def agent(user_input, max_steps=20):
    # max_steps 防止无限调用

    # 首次调用 LLM 的 payload（中的 messages）：
    messages = [
        {"role": "system", "content": "You are an intelligent assistant. Use tools when necessary."},
        {"role": "user", "content": user_input}
    ]

    for _ in range(max_steps):
        payload = {
            "model": model_name,
            "messages": messages,
            # 告诉 LLM 有哪些工具可以调用：
            "tools": tools,
            "tool_choice": "auto"
        }

        resp = call_llm.call(payload)
        message = resp["choices"][0]["message"]

        # 模型的回复添加到 Context
        # 对于生产级别的 agent，这里还需要 Context 的压缩等
        messages.append(message)

        # 任务完成条件：没有 tool_calls，说明模型认为任务已经完成
        if not message.get("tool_calls"):
            return message["content"]

        # 执行工具
        for tool_call in message["tool_calls"]:
            result = tool_routine(tool_call)

            # 4. 将工具结果加入 Context
            messages.append({
                "role": "tool",
                "tool_call_id": tool_call["id"],
                "content": result
            })

    return "Agent exceeded max_steps."
```

### 上下文压缩


上面的 Loop 能持续运行，但是 Request 不应该无限增长，尤其是已经无用的信息，否则每次调用的 token 预算持续增长，费用也会越来越多，最终超过 LLM 的窗口上限。


Context 中有哪些内容？

| 名称 | 内容 | 做法 |
|-----|------|-----|
| **System Policy** | 角色、权限、事实边界、冲突规则 | 始终注入、不可篡改、不可压缩
| **Context Policy** 上下文策略 | Context 压缩策略，block 顺序、触发压缩的阈值、Tool 截断等信息 | 用于 Harness，不进入 Context
| **Task / Runtime** 任务状态 ｜ 目标、约束、计划、当前执行进度 | 按需注入
| **Evidence Ledger** 事实账本 | 关键事实、状态、来源 ID | 仅注入 Active
| **History/Trace** 历史与轨迹 | 历史对话、工具调用、执行事件 | 按需检索或重放
| **Skills / MCP** 能力与工具 | 技能说明、工具 Schema | 渐进加载、相关时才加载
| **Raw Results / Compaction Log** 原始与压缩日志 | 原始工具结果、旧摘要、压缩过程 | 不默认交给主模型
